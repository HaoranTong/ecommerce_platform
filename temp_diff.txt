+    def _entity_needs_data_chain(self, entity_type: str) -> bool:
+        """自动判断实体是否需要完整数据链（使用配置文件，避免硬编码）
+        
+        判断策略（按优先级）：
+        1. 检查cross_module_dependency_chains配置（有依赖链的实体需要完整数据链）
+        2. 检查common_models配置（判断实体所属模块）
+        3. product_catalog模块的所有实体互相依赖，需要完整数据链
+        
+        Args:
+            entity_type: 实体类型名称（如'product', 'sku', 'order'）
+            
+        Returns:
+            bool: True表示需要完整数据链，False表示可以单独创建
+        """
+        # 转换为PascalCase模型名（entity_type通常是小写的）
+        model_name = entity_type.capitalize()
+        if entity_type.lower() == 'sku':
+            model_name = 'SKU'
+        
+        # 1. 检查配置文件中的依赖链
+        dependency_chains = self.config.get('business_logic_patterns', {}).get('cross_module_dependency_chains', {})
+        if model_name in dependency_chains:
+            # 如果配置中有依赖链，说明是复杂实体
+            return True
+        
+        # 2. 检查实体所属模块
+        common_models = self.config.get('business_logic_patterns', {}).get('common_models', {})
+        entity_module = common_models.get(model_name, [])
+        
+        # 3. product_catalog模块的实体互相依赖
+        if 'product_catalog' in entity_module or entity_type in ['product', 'sku', 'category', 'brand', 'inventory']:
+            return True
+        
+        # 4. 其他实体默认可以单独创建
+        return False
